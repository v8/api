1000// static
1001String::ExternalStringResourceBase* String::ReadExternalStringResource(
1002 v8::Isolate* isolate, internal::Address obj) {
1003 using A = internal::Address;
1004 using I = internal::Internals;
1005 // The resource field references a cppgc wrapper object (through the
1006 // CppHeapPointerTable when pointer compression is enabled), which in turn
1007 // holds the resource. The wrapper is nullptr if the type check fails.
1008 A wrapper = reinterpret_cast<A>(internal::ReadCppHeapPointerField<void>(
1009 isolate, obj, I::kStringResourceOffset,
1010 CppHeapPointerTag::kExternalStringResourceTag));
1011 if (wrapper == internal::kNullAddress) return nullptr;
1012 return *reinterpret_cast<ExternalStringResourceBase* const*>(
1013 wrapper + I::kExternalStringResourceWrapperResourceOffset);
1014}

1016String::ExternalStringResource* String::GetExternalStringResource() const {
1017 using A = internal::Address;
1018 using I = internal::Internals;
1019 A obj = internal::ValueHelper::ValueAsAddress(this);
1020
1021 ExternalStringResource* result;
1022 if (I::IsExternalTwoByteString(I::GetInstanceType(obj))) {
1023 Isolate* isolate = I::GetCurrentIsolate();
1024 result = static_cast<ExternalStringResource*>(
1025 ReadExternalStringResource(isolate, obj));
1026 } else {
1027 result = GetExternalStringResourceSlow();
1028 }
1029#ifdef V8_ENABLE_CHECKS
1030 VerifyExternalStringResourceBase(result, Encoding::TWO_BYTE_ENCODING);
1031#endif
1032 return result;
1033}

1054String::ExternalStringResourceBase* String::GetExternalStringResourceBase(
1055 v8::Isolate* isolate, String::Encoding* encoding_out) const {
1056 using A = internal::Address;
1057 using I = internal::Internals;
1058 A obj = internal::ValueHelper::ValueAsAddress(this);
1059 int type = I::GetInstanceType(obj) & I::kStringRepresentationAndEncodingMask;
1060 *encoding_out = static_cast<Encoding>(type & I::kStringEncodingMask);
1061 ExternalStringResourceBase* resource;
1062 if (type == I::kExternalOneByteRepresentationTag ||
1063 type == I::kExternalTwoByteRepresentationTag) {
1064 resource = ReadExternalStringResource(isolate, obj);
1065 } else {
1066 resource = GetExternalStringResourceBaseSlow(encoding_out);
1067 }
1068#ifdef V8_ENABLE_CHECKS
1069 VerifyExternalStringResourceBase(resource, *encoding_out);
1070#endif
1071 return resource;
1072}

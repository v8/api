44struct JitCodeEvent {
45 enum EventType {
46 CODE_ADDED,
47 CODE_MOVED,
48 CODE_REMOVED,
49 CODE_ADD_LINE_POS_INFO,
50 CODE_START_LINE_INFO_RECORDING,
51 CODE_END_LINE_INFO_RECORDING
52 };
53 // Definition of the code position type. The "POSITION" type means the place
54 // in the source code which are of interest when making stack traces to
55 // pin-point the source location of a stack frame as close as possible.
56 // The "STATEMENT_POSITION" means the place at the beginning of each
57 // statement, and is used to indicate possible break locations.
58 enum PositionType { POSITION, STATEMENT_POSITION };
59
60 // There are three different kinds of CodeType, one for JIT code generated
61 // by the optimizing compiler, one for byte code generated for the
62 // interpreter, and one for code generated from Wasm. For JIT_CODE and
63 // WASM_CODE, |code_start| points to the beginning of jitted assembly code,
64 // while for BYTE_CODE events, |code_start| points to the first bytecode of
65 // the interpreted function.
66 enum CodeType { BYTE_CODE, JIT_CODE, WASM_CODE };
67
68 // Type of event.
69 EventType type;
70 CodeType code_type;
71 // Start of the instructions.
72 void* code_start;
73 // Size of the instructions.
74 size_t code_len;
75 // Script info for CODE_ADDED event.
76 Local<UnboundScript> script;
77 // User-defined data for *_LINE_INFO_* event. It's used to hold the source
78 // code line information which is returned from the
79 // CODE_START_LINE_INFO_RECORDING event. And it's passed to subsequent
80 // CODE_ADD_LINE_POS_INFO and CODE_END_LINE_INFO_RECORDING events.
81 void* user_data;
82
83 struct name_t {
84 // Name of the object associated with the code, note that the string is not
85 // zero-terminated.
86 const char* str;
87 // Number of chars in str.
88 size_t len;
89 };
90
91 struct line_info_t {
92 // PC offset
93 size_t offset;
94 // Code position
95 size_t pos;
96 // The position type.
97 PositionType position_type;
98 };
99
100 struct wasm_source_info_t {
101 // Source file name.
102 const char* filename;
103 // Length of filename.
104 size_t filename_size;
105 // Line number table, which maps offsets of JITted code to line numbers of
106 // source file.
107 const line_info_t* line_number_table;
108 // Number of entries in the line number table.
109 size_t line_number_table_size;
110 };
111
112 wasm_source_info_t* wasm_source_info = nullptr;
113
114 union {
115 // Only valid for CODE_ADDED.
116 struct name_t name;
117
118 // Only valid for CODE_ADD_LINE_POS_INFO
119 struct line_info_t line_info;
120
121 // New location of instructions. Only valid for CODE_MOVED.
122 void* new_code_start;
123 };
124
125 Isolate* isolate;
126};

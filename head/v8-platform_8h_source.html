328// Allows a thread to temporarily boost another thread's priority to match its

339 // Boosts the priority of the thread where this ScopedBoostablePriority was
340 // created. Can be called from any thread, but requires proper external

481// Opaque type representing a handle to a shared memory region.
482class SharedMemoryHandle {
483 public:
484 // For the handle itself, we use the underlying type (e.g. unsigned int)
485 // instead of e.g. mach_port_t to avoid pulling in large OS header files into
486 // this header file. Instead, the users of these routines are expected to
487 // include the respective OS headers in addition to this one.
488
489#if V8_OS_DARWIN
490 // A mach_port_t referencing a memory entry object.
491 using PlatformHandle = unsigned int;
492#elif V8_OS_FUCHSIA
493 // A zx_handle_t to a VMO.
494 using PlatformHandle = uint32_t;
495#elif V8_OS_WIN
496 // A Windows HANDLE to a file mapping object.
497 using PlatformHandle = void*;
498#else
499 // A file descriptor.
500 using PlatformHandle = int;
501#endif
502
503 static constexpr SharedMemoryHandle FromPlatformHandle(
504 PlatformHandle handle) {
505 return SharedMemoryHandle(handle);
506 }
507
508 PlatformHandle GetPlatformHandle() const { return handle_; }
509
510 private:
511 SharedMemoryHandle() = delete;
512 explicit constexpr SharedMemoryHandle(PlatformHandle handle)
513 : handle_(handle) {}
514
515 PlatformHandle handle_;
516};

868inline constexpr PagePermissions operator|(PagePermissions lhs,

877 static_cast<int>(rhs));

1153 /*
1154 * Allocate a subspace.
1155 *
1156 * The address space of a subspace stays reserved in the parent space for the
1157 * lifetime of the subspace. As such, it is guaranteed that page allocations
1158 * on the parent space cannot end up inside a subspace.
1159 *
1160 * \param hint Hints where the subspace should be allocated. See
1161 * AllocatePages() for more details.
1162 *
1163 * \param size The size in bytes of the subspace. Must be a multiple of the
1164 * allocation_granularity().
1165 *
1166 * \param alignment The alignment of the subspace in bytes. Must be a multiple
1167 * of the allocation_granularity() and should be a power of two.
1168 *
1169 * \param max_page_permissions The maximum permissions that pages allocated in
1170 * the subspace can obtain.
1171 *
1172 * \param key Optional memory protection key for the subspace. If used, the
1173 * returned subspace will use this key for all its memory pages.
1174 *
1175 * \param handle Optional file descriptor for the subspace. If used, the
1176 * returned subspace will use this file descriptor with 0 offset as the
1177 * space's underlying file.
1178 *
1179 * \returns a new subspace or nullptr on failure.
1180 */

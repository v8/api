35struct OwnedBuffer {
36 V8_DEPRECATE_SOON("Use as_span() instead")
37 std::unique_ptr<const uint8_t[]> buffer;
38 V8_DEPRECATE_SOON("Use as_span() instead")
39 size_t size = 0;
40
41 START_ALLOW_USE_DEPRECATED()
42 OwnedBuffer() = default;
43 ~OwnedBuffer() = default;
44
45 V8_DEPRECATE_SOON(
46 "Use as_span() instead; external construction is deprecated")
47 OwnedBuffer(std::unique_ptr<const uint8_t[]> buffer, size_t size)
48 : buffer(std::move(buffer)), size(size) {}
49
50 OwnedBuffer(const OwnedBuffer&) = delete;
51 OwnedBuffer& operator=(const OwnedBuffer&) = delete;
52
53 OwnedBuffer(OwnedBuffer&& other) noexcept
54 : buffer(std::move(other.buffer)), size(std::exchange(other.size, 0)) {}
55
56 OwnedBuffer& operator=(OwnedBuffer&& other) noexcept {
57 if (this != &other) {
58 buffer = std::move(other.buffer);
59 size = std::exchange(other.size, 0);
60 }
61 return *this;
62 }
63
64 constexpr bool empty() const { return size == 0; }
65
66 constexpr std::span<const uint8_t> as_span() const {
67 return {buffer.get(), size};
68 }
69 END_ALLOW_USE_DEPRECATED()
70};

152 struct CompileOptions {
153 // Builtin compile-time imports, mirroring the strings accepted in the
154 // `builtins` array of the JS `WebAssembly.Module` constructor options.
155 // Combine values with bitwise-or to enable multiple builtins.
156 struct Builtins {
157 enum {
158 kNone = 0,
159 kJsString = 1 << 0, // "js-string"
160 };
161 };
162 // Bitwise-or of `Builtins` values to enable as compile-time imports.
163 int builtins = Builtins::kNone;
164 // If non-null, enable imported string constants from the named module
165 // (e.g. "wasm:js/string-constants"). The string must be null-terminated and
166 // remain valid for the duration of the compile call.

168 // If non-empty, associated with the module's script as its source URL, for

34struct OwnedBuffer {
35 OwnedBuffer() = default;
36 ~OwnedBuffer() = default;
37
38 OwnedBuffer(const OwnedBuffer&) = delete;
39 OwnedBuffer& operator=(const OwnedBuffer&) = delete;
40
41 OwnedBuffer(OwnedBuffer&& other) noexcept
42 : buffer_(std::move(other.buffer_)),
43 size_(std::exchange(other.size_, 0)) {}
44
45 OwnedBuffer& operator=(OwnedBuffer&& other) noexcept {
46 if (this != &other) {
47 buffer_ = std::move(other.buffer_);
48 size_ = std::exchange(other.size_, 0);
49 }
50 return *this;
51 }
52
53 constexpr bool empty() const { return size_ == 0; }
54
55 constexpr std::span<const uint8_t> as_span() const {
56 return {buffer_.get(), size_};
57 }
58
59 private:
60 friend class CompiledWasmModule;
61
62 OwnedBuffer(std::unique_ptr<const uint8_t[]> buffer, size_t size)
63 : buffer_(std::move(buffer)), size_(size) {}
64
65 std::unique_ptr<const uint8_t[]> buffer_;
66 size_t size_ = 0;
67};

974Local<Data> Object::GetInternalField(int index) {
975#ifndef V8_ENABLE_CHECKS
976 using A = internal::Address;
977 using I = internal::Internals;
978 A obj = internal::ValueHelper::ValueAsAddress(this);
979 // Fast path: If the object is a plain JSObject, which is the common case, we
980 // know where to find the internal fields and can return the value directly.
981 int instance_type = I::GetInstanceType(obj);
982 if (I::CanHaveInternalField(instance_type)) {
983 int offset = I::kJSAPIObjectWithEmbedderSlotsHeaderSize +
984 (I::kEmbedderDataSlotSize * index);
985#ifdef V8_COMPRESS_POINTERS
986 // The tagged payload lives in the low kTaggedSize half of the slot (at
987 // kTaggedPayloadOffset == 0). Read it as a 32-bit field so the correct half
988 // is picked on both little and big endian targets. A full width read plus
989 // truncation would return the CppHeap pointer half on big endian.
990 uint32_t compressed = I::ReadRawField<uint32_t>(obj, offset);
991 A value = I::DecompressTaggedField(obj, compressed);
992#else
993 A value = I::ReadRawField<A>(obj, offset);
994#endif
995
996 auto* isolate = I::GetCurrentIsolate();
997 return Local<Data>::New(isolate, value);
998 }
999#endif
1000 return SlowGetInternalField(index);
1001}

1014template <typename T>
1015 requires cppgc::IsGarbageCollectedTypeV<T>
1016T* Object::GetPointerFromInternalField(v8::Isolate* isolate, int index,
1017 CppHeapPointerTag tag) {
1018#if !defined(V8_ENABLE_CHECKS)
1019 using A = internal::Address;
1020 using I = internal::Internals;
1021 A obj = internal::ValueHelper::ValueAsAddress(this);
1022 // Fast path: If the object is a plain JSObject, which is the common case, we
1023 // know where to find the internal fields and can return the value directly.
1024 auto instance_type = I::GetInstanceType(obj);
1025 if (V8_LIKELY(I::CanHaveInternalField(instance_type))) {
1026 int offset = I::kJSAPIObjectWithEmbedderSlotsHeaderSize +
1027 (I::kEmbedderDataSlotSize * index) +
1028 I::kEmbedderDataSlotCppHeapPointerOffset;
1029 return internal::ReadCppHeapPointerField<T>(
1030 isolate, obj, offset, CppHeapPointerTagRange(tag, tag));
1031 }
1032#endif
1033 return static_cast<T*>(SlowGetPointerFromInternalField(isolate, index, tag));
1034}
